CREATE TABLE applicants (
    -- Auto-incrementing ID for new user registrations
    id SERIAL PRIMARY KEY,
    
    -- Authentication (Since this is for registration)
    password_hash VARCHAR(255), 
    
    -- ==========================================
    -- C1. PERSONAL INFORMATION
    -- ==========================================
    surname VARCHAR(100) NOT NULL,
    first_name VARCHAR(100) NOT NULL,
    middle_name VARCHAR(100),
    date_of_birth DATE,
    place_of_birth VARCHAR(255),
    sex VARCHAR(20), 
    civil_status VARCHAR(50), 
    citizenship VARCHAR(100) DEFAULT 'Filipino',
    blood_type VARCHAR(50),
    religion VARCHAR(100),
    disability VARCHAR(100),
    ethnic_group VARCHAR(100),
    
    -- Government IDs
    gsis_id_no VARCHAR(50),
    pag_ibig_id_no VARCHAR(50),
    philhealth_no VARCHAR(50),
    sss_no VARCHAR(50),
    
    -- Contact Information
    residential_address TEXT,
    permanent_address TEXT,
    telephone_no VARCHAR(50),
    mobile_no VARCHAR(50),
    email_address VARCHAR(150) UNIQUE NOT NULL,

    -- ==========================================
    -- C2. FAMILY BACKGROUND
    
    -- ==========================================
    spouse_surname VARCHAR(100),
    spouse_first_name VARCHAR(100),
    spouse_occupation VARCHAR(150),
    
    father_surname VARCHAR(100),
    father_first_name VARCHAR(100),
    mother_maiden_surname VARCHAR(100),

    -- ==========================================
    -- TABULAR DATA (Stored as JSONB arrays)
    -- ==========================================
    
    -- Stores array of objects: { level, school_name, degree, date_from, date_to, highest_level, year_graduated }
    educational_background JSONB DEFAULT '[]'::jsonb, 
    
    -- Stores array of objects: { eligibility, rating, date_of_exam, place_of_exam, license_no, license_validity }
    civil_service_eligibility JSONB DEFAULT '[]'::jsonb,
    
    -- Stores array of objects: { date_from, date_to, position_title, agency_company, monthly_salary, pay_grade, status, govt_service }
    work_experience JSONB DEFAULT '[]'::jsonb,
    
    -- Stores array of objects: { organization_name, date_from, date_to, hours, position }
    voluntary_work JSONB DEFAULT '[]'::jsonb,
    
    -- Stores array of objects: { title, date_from, date_to, hours, type_of_ld, sponsor }
    learning_and_development JSONB DEFAULT '[]'::jsonb,
    
    -- Stores object: { skills: [], distinctions: [], organizations: [] }
    other_information JSONB DEFAULT '{}'::jsonb,

    -- ==========================================
    -- C4. QUESTIONNAIRE (Stored as JSONB)
    -- ==========================================
    -- Example structure: 
    -- {
    --    "q34a": { "answer": "no", "details": "" },
    --    "q34b": { "answer": "yes", "details": "Mayor John Doe" }
    -- }
    questionnaire_responses JSONB DEFAULT '{}'::jsonb,

    -- ==========================================
    -- TIMESTAMPS & TRACKING
    -- ==========================================
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE saved_jobs (
    id SERIAL PRIMARY KEY,
    applicant_id INTEGER NOT NULL REFERENCES applicants(id) ON DELETE CASCADE,
    position_id INTEGER NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(applicant_id, position_id)
